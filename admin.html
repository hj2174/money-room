📜 Supabase DB 테이블 생성 SQL 가이드

Supabase SQL Editor에 아래 쿼리를 실행하시면 furniture_catalog 테이블 및 권한 정책이 자동으로 생성됩니다. 
-- 1. 가구 카탈로그 마스터 테이블 생성
CREATE TABLE IF NOT EXISTS furniture_catalog (
    id VARCHAR(100) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL DEFAULT 'bed',
    base_w INT NOT NULL DEFAULT 1,
    base_h INT NOT NULL DEFAULT 1,
    price INT NOT NULL DEFAULT 0,
    description TEXT,
    image_url TEXT,
    svg_data TEXT,
    is_special BOOLEAN DEFAULT FALSE,
    condition_type VARCHAR(50),
    condition_value BIGINT DEFAULT 0,
    condition_desc TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. RLS 활성화 및 공용 접근 정책 (조회/생성/수정/삭제)
ALTER TABLE furniture_catalog ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public Read Furniture Catalog" ON furniture_catalog FOR SELECT USING (true);
CREATE POLICY "Public Insert Furniture Catalog" ON furniture_catalog FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Update Furniture Catalog" ON furniture_catalog FOR UPDATE USING (true);
CREATE POLICY "Public Delete Furniture Catalog" ON furniture_catalog FOR DELETE USING (true);

-- 3. 실시간 동기화 복제 활성화 (선택)
ALTER PUBLICATION supabase_realtime ADD TABLE furniture_catalog;
